cmake_minimum_required(VERSION 3.20)
project(aether_execution_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
if(TARGET protobuf::protoc)
  set(PROTOC "$<TARGET_FILE:protobuf::protoc>")
else()
  set(PROTOC "${Protobuf_PROTOC_EXECUTABLE}")
endif()
set(GRPC_CPP_PLUGIN "$<TARGET_FILE:gRPC::grpc_cpp_plugin>")
set(PROTO "${CMAKE_CURRENT_SOURCE_DIR}/../../proto/execution.proto")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
add_custom_command(
  OUTPUT "${GEN}/execution.pb.cc" "${GEN}/execution.pb.h" "${GEN}/execution.grpc.pb.cc" "${GEN}/execution.grpc.pb.h"
  COMMAND "${PROTOC}" --proto_path="${CMAKE_CURRENT_SOURCE_DIR}/../../proto" --cpp_out="${GEN}" --grpc_out="${GEN}" --plugin=protoc-gen-grpc="${GRPC_CPP_PLUGIN}" "${PROTO}"
  DEPENDS "${PROTO}"
  VERBATIM)
add_library(aether_proto "${GEN}/execution.pb.cc" "${GEN}/execution.grpc.pb.cc")
target_include_directories(aether_proto PUBLIC "${GEN}")
target_link_libraries(aether_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(aether_compute src/compute.cpp)
target_link_libraries(aether_compute PUBLIC OpenSSL::Crypto)
add_executable(aether-execution-engine src/main.cpp)
target_link_libraries(aether-execution-engine PRIVATE aether_proto aether_compute Threads::Threads)
include(CTest)
if(BUILD_TESTING)
 add_executable(compute_test tests/compute_test.cpp)
 target_link_libraries(compute_test PRIVATE aether_compute)
 add_test(NAME compute_test COMMAND compute_test)
endif()
